75static void dht_poll_task(void *arg)
76{
77 (void)arg;
78
79 const TickType_t delay_ticks = pdMS_TO_TICKS(APP_DHT22_POLL_INTERVAL_MS);
80
81 // Initialize with invalid sample
82 latest_sample_t s = {
83 .ts_us = esp_timer_get_time(),
84 .temp_c = 0,
85 .rh = 0,
86 .valid = 0,
87 .err = DHT22_ERR_TIMEOUT,
88 .seq = 0,
89 };
90 sample_set(&s);
91
92 while (1) {
93 int64_t now = esp_timer_get_time();
94
95 dht22_reading_t r;
96 dht22_err_t e = dht22_read(APP_DHT22_GPIO, &r, 5000);
97
98 s.ts_us = now;
99 s.seq++;
100 if (e == DHT22_OK) {
101 s.temp_c = r.temperature_c;
102 s.rh = r.humidity_rh;
103 s.valid = 1;
104 s.err = 0;
105 } else {
106 s.valid = 0;
107 s.err = (int)e;
108 }
109 sample_set(&s);
110
111 vTaskDelay(delay_ticks);
112 }

115static void dht_stream_task(void *arg)
116{
117 (void)arg;
118
119#if APP_PRINT_CSV_HEADER
120 const char *hdr = "ts_us,temp_c,humidity_rh,valid,err,seq,age_ms\r\n";
121 (void)uart_write_bytes(APP_UART_NUM, hdr, (size_t)strlen(hdr));
122#endif
123
124 const TickType_t delay_ticks = pdMS_TO_TICKS(APP_STREAM_PERIOD_MS);
125
126 while (1) {
127 const int64_t now = esp_timer_get_time();
128 latest_sample_t s = sample_get();
129 int64_t age_us = now - s.ts_us;
130 if (age_us < 0) {
131 age_us = 0;
132 }
133
134 char line[128];
135 int n = snprintf(line, sizeof(line),
136 "%" PRId64 ",%.1f,%.1f,%d,%d,%" PRIu32 ",%" PRId64 "\r\n",
137 now,
138 (double)s.temp_c,
139 (double)s.rh,
140 s.valid,
141 s.err,
142 s.seq,
143 (int64_t)(age_us / 1000));
144
145 if (n > 0) {
146 int wr = uart_write_bytes(APP_UART_NUM, line, (size_t)n);
147 if (wr < 0) {
148 s_dropped_lines++;
149 }
150 } else {
151 s_dropped_lines++;
152 }
153
154 vTaskDelay(delay_ticks);
155 }

158void dht22_streamer_start(void)
159{
160 uart0_init_once();
161
162 if (!s_lock) {
163 s_lock = xSemaphoreCreateMutex();
164 }
165
166 // Keep tasks pinned to no core (ESP32-C3 is single core)
167 xTaskCreate(dht_poll_task, "dht_poll", 4096, NULL, 8, NULL);
168 xTaskCreate(dht_stream_task, "dht_stream", 4096, NULL, 7, NULL);
169
170 ESP_LOGI(TAG, "DHT22 streamer started (GPIO=%d poll=%dms stream=%dms baud=%d)",
171 APP_DHT22_GPIO, APP_DHT22_POLL_INTERVAL_MS, APP_STREAM_PERIOD_MS, APP_UART_BAUD);
172}
